# Test registration. Every GoogleTest binary links the harness in test/support,
# which reduces failure status across the communicator (docs/dev/testing.md).
# Tests that still carry their upstream custom main() -- the rename-only state
# of a port -- are registered as plain executables until their conversion
# commit lands. There is no third kind.

add_library(fuse_test_main STATIC support/fuse_test_main.cpp)
target_include_directories(fuse_test_main PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/support)
target_link_libraries(fuse_test_main PUBLIC fuse::base GTest::gtest)
if(FUSE_ENABLE_MPI)
  target_link_libraries(fuse_test_main PUBLIC MPI::MPI_CXX)
endif()

# fuse_add_gtest(<layer> <name> [LIBS ...]) -> test/<layer>/<name>_test.cpp
# THROWS marks a test that asserts an exception is raised; the no-exceptions CI
# leg excludes that label because with FUSE_ENABLE_EXCEPTIONS=OFF the library
# aborts instead, by contract. Membership is declared here, not detected --
# a parent tried detecting it by grep, in both directions, and both were wrong.
# Golden files are data the tests compare against, never regenerated by them.
set(FUSE_TEST_GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/golden")
set(FUSE_TEST_DATA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/data")

# `layer` names both the source directory and, by default, the library the test
# links. NO_LAYER_LIB separates them: test/verification/ is not a library layer,
# it exercises whatever the case needs (declared via LIBS).
# Every registration function records its source in FUSE_REGISTERED_TEST_SOURCES;
# the audit at the bottom of this file compares that against the files on disk.
# A test file that exists but is registered nowhere runs nowhere, and nothing
# else reports it: 33 cases in schedule/buffers_test.cpp and slab_test.cpp sat
# unregistered for four days (REVIEW.md R-10).
set_property(GLOBAL PROPERTY FUSE_REGISTERED_TEST_SOURCES "")
set_property(GLOBAL PROPERTY FUSE_GATED_TEST_SOURCES "")
function(fuse_record_test_source rel)
  set_property(GLOBAL APPEND PROPERTY FUSE_REGISTERED_TEST_SOURCES "${rel}")
endfunction()
# fuse_tests_gated_by(<option> <rel>...) -- these sources are legitimately
# unregistered when <option> is OFF (absent, not skipped: a skipped test reads
# like a passing one). Called UNconditionally so the audit knows about them in
# every configuration.
function(fuse_tests_gated_by option)
  if(NOT ${option})
    foreach(rel IN LISTS ARGN)
      set_property(GLOBAL APPEND PROPERTY FUSE_GATED_TEST_SOURCES "${rel}")
    endforeach()
  endif()
endfunction()

function(fuse_add_gtest layer name)
  cmake_parse_arguments(ARG "THROWS;NO_LAYER_LIB" "" "LIBS" ${ARGN})
  set(target fuse_${layer}_${name}_test)
  add_executable(${target} ${layer}/${name}_test.cpp)
  fuse_record_test_source(${layer}/${name}_test.cpp)
  if(ARG_NO_LAYER_LIB)
    target_link_libraries(${target} PRIVATE fuse_test_main ${ARG_LIBS})
  else()
    target_link_libraries(${target} PRIVATE fuse::${layer} fuse_test_main ${ARG_LIBS})
  endif()
  target_compile_definitions(${target} PRIVATE FUSE_TEST_GOLDEN_DIR="${FUSE_TEST_GOLDEN_DIR}"
                                             FUSE_TEST_DATA_DIR="${FUSE_TEST_DATA_DIR}")
  add_test(NAME ${layer}.${name} COMMAND ${target})
  set(labels "${layer};gtest")
  if(ARG_THROWS)
    list(APPEND labels throws)
  endif()
  set_tests_properties(${layer}.${name} PROPERTIES LABELS "${labels}")
endfunction()

# fuse_add_upstream_main_test(<layer> <name>) -> same path, own main(), no harness.
function(fuse_add_upstream_main_test layer name)
  set(target fuse_${layer}_${name}_test)
  add_executable(${target} ${layer}/${name}_test.cpp)
  fuse_record_test_source(${layer}/${name}_test.cpp)
  target_link_libraries(${target} PRIVATE fuse::${layer})
  add_test(NAME ${layer}.${name} COMMAND ${target})
  set_tests_properties(${layer}.${name} PROPERTIES LABELS "${layer};upstream-main")
endfunction()

# fuse_add_mpi_test(<layer> <name> [LIBS ...] [RANKS n...]) -> mpiexec at each
# rank count. Default ranks 2 3 4 5: an odd count is included on purpose, since
# a bug that appears only when the ranks do not divide evenly is the usual kind.
# Every leg carries a timeout, not because these are slow: the failure they
# guard against IS a hang, so a regression must fail rather than stall CI.
function(fuse_add_mpi_test layer name)
  cmake_parse_arguments(ARG "" "" "LIBS;RANKS" ${ARGN})
  if(NOT ARG_RANKS)
    set(ARG_RANKS 2 3 4 5)
  endif()
  set(target fuse_${layer}_${name}_test)
  add_executable(${target} ${layer}/${name}_test.cpp)
  fuse_record_test_source(${layer}/${name}_test.cpp)
  target_link_libraries(${target} PRIVATE fuse_test_main ${ARG_LIBS})
  target_compile_definitions(${target} PRIVATE FUSE_TEST_GOLDEN_DIR="${FUSE_TEST_GOLDEN_DIR}"
                                             FUSE_TEST_DATA_DIR="${FUSE_TEST_DATA_DIR}")
  foreach(n IN LISTS ARG_RANKS)
    add_test(NAME ${layer}.${name}.${n}
             COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${n} $<TARGET_FILE:${target}>)
    set_tests_properties(${layer}.${name}.${n} PROPERTIES TIMEOUT 120 LABELS "${layer};mpi;parallel")
  endforeach()
endfunction()

# ---------------------------------------------------------------- ir
# THROWS below is not decoration: these targets contain cases that assert a
# refusal is raised, and with FUSE_ENABLE_EXCEPTIONS=OFF a refusal aborts, so
# the whole target would abort in the no-exceptions leg. The label is
# target-granular, so excluding one also costs the coverage of its
# non-throwing cases -- an accepted price for now, and the reason to prefer
# per-case guards if this list grows. Determined by running the leg, not by
# reading: the nine here are exactly the targets that still aborted once the
# library stopped using throw/catch as internal control flow.

# A rule enforced by a DELETED OVERLOAD can only be tested by failing to
# compile. Both targets are EXCLUDE_FROM_ALL and built only by their test, so a
# normal build is unaffected.
#
# The pair matters: WILL_FAIL alone would pass if the file failed to build for
# any reason at all -- a typo, a missing include -- so the control is the same
# shape written the permitted way and MUST build. Together they say the
# difference is the spelling.
foreach(case compile_fail_dangling_span compile_ok_owning_chain)
  add_executable(fuse_field_${case} EXCLUDE_FROM_ALL field/${case}.cpp)
  target_link_libraries(fuse_field_${case} PRIVATE fuse::base)
endforeach()

add_test(NAME field.compile_fail.dangling_span
  COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR}
          --target fuse_field_compile_fail_dangling_span)
set_tests_properties(field.compile_fail.dangling_span PROPERTIES
  WILL_FAIL TRUE LABELS "compile_fail")

add_test(NAME field.compile_ok.owning_chain
  COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR}
          --target fuse_field_compile_ok_owning_chain)
set_tests_properties(field.compile_ok.owning_chain PROPERTIES
  LABELS "compile_fail")

# ---------------------------------------------------------------- install
# find_package(fuse-coupler) smoke test (REVIEW.md R-25/BT-F9): install this
# build, configure and build a tiny STANDALONE consumer project against the
# install prefix (test/install_consumer/, its own independent CMake project,
# never add_subdirectory'd into this one), then run it. A `cmake -P` script
# drives the three cmake invocations because ctest's COMMAND runs one program.
#
# This is the actual thing conventions.md's "CMake package `fuse-coupler`"
# claim needs to be true, not a proxy for it: before this, nothing in the
# tree ran `cmake --install` at all.
add_test(NAME install.consumer_find_package
  # -D arguments MUST precede -P: cmake only honours -D definitions that
  # appear before the script argument on its command line.
  COMMAND ${CMAKE_COMMAND}
          -DFUSE_BUILD_DIR=${CMAKE_BINARY_DIR}
          -DFUSE_INSTALL_PREFIX=${CMAKE_BINARY_DIR}/install-test
          -DCONSUMER_SRC_DIR=${CMAKE_CURRENT_SOURCE_DIR}/install_consumer
          -DCONSUMER_BUILD_DIR=${CMAKE_BINARY_DIR}/install_consumer_build
          -DCMAKE_COMMAND=${CMAKE_COMMAND}
          -DCMAKE_GENERATOR=${CMAKE_GENERATOR}
          "-DCMAKE_PREFIX_PATH=${CMAKE_PREFIX_PATH}"
          -P ${CMAKE_CURRENT_SOURCE_DIR}/install_consumer/run-smoke-test.cmake
)
set_tests_properties(install.consumer_find_package PROPERTIES
  LABELS "install"
  # Builds a second small project from scratch; generous relative to the rest
  # of the suite (docs/dev/testing.md's timeout convention for anything that
  # runs more than one build).
  TIMEOUT 300
)

# ---------------------------------------------------------------- audit
# Every *_test.cpp on disk must be registered above or declared gated by a tier
# option that is OFF in this configuration. Anything else is a test that runs
# nowhere, and this is the only place that would ever say so.
file(GLOB_RECURSE _fuse_test_files RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
     ${CMAKE_CURRENT_SOURCE_DIR}/*/*_test.cpp)
get_property(_fuse_registered GLOBAL PROPERTY FUSE_REGISTERED_TEST_SOURCES)
get_property(_fuse_gated GLOBAL PROPERTY FUSE_GATED_TEST_SOURCES)
set(_fuse_unregistered "")
foreach(f IN LISTS _fuse_test_files)
  if(NOT f IN_LIST _fuse_registered AND NOT f IN_LIST _fuse_gated)
    list(APPEND _fuse_unregistered ${f})
  endif()
endforeach()
if(_fuse_unregistered)
  list(JOIN _fuse_unregistered "\n    " _fuse_unregistered_text)
  message(FATAL_ERROR "fuse: test source(s) registered nowhere, so they would run nowhere:\n"
                      "    ${_fuse_unregistered_text}\n"
                      "  Register each with fuse_add_gtest / fuse_add_mpi_test, or declare it "
                      "gated with fuse_tests_gated_by(<option> <file>).")
endif()
list(LENGTH _fuse_test_files _n_files)
list(LENGTH _fuse_registered _n_reg)
message(STATUS "fuse: tests  ${_n_reg} registered of ${_n_files} on disk (rest gated by a tier option that is OFF)")
