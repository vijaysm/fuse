# A handful of tests below (runtime/field .compile_ok/.compile_fail) drive a
# NESTED `cmake --build ${CMAKE_BINARY_DIR} --target ...` at test-run time
# rather than linking a normal gtest binary, specifically so the thing under
# test is "does this translation unit compile," not "does it run." That
# nested build re-invokes mpicxx, which is a shell wrapper that picks its
# real host compiler from $OMPI_CXX at the moment IT runs -- not from
# whatever was exported when this project was configured. If OMPI_CXX
# (/opt/kokkos/bin/nvcc_wrapper on a Kokkos::CUDA build, set by every
# CMakePresets.json tier) is missing from the shell that happens to invoke
# ctest, mpicxx silently falls back to plain g++, which rejects the
# nvcc-only flags (`-Wext-lambda-captures-this`, `-arch=sm_61`) this
# project's own CMake unconditionally adds whenever Kokkos::CUDA exists --
# breaking runtime.compile_ok.capabilities_override and
# field.compile_ok.owning_chain outright, and silently turning their
# WILL_FAIL siblings into false passes (a bad build fails for the wrong
# reason instead of the one under test). Pinning OMPI_CXX to what THIS
# configure actually used, the same way MOAB's LD_LIBRARY_PATH fixup above
# does, makes the nested build's toolchain match this one regardless of the
# invoking shell.
set(_fuse_nested_build_env_mods "")
if(DEFINED ENV{OMPI_CXX})
  list(APPEND _fuse_nested_build_env_mods "OMPI_CXX=set:$ENV{OMPI_CXX}")
endif()

# ---- the OpenMP thread-count leg -------------------------------------------
# A Kokkos built with OpenMP runs every test on OpenMP already, at whatever
# thread count the machine and environment happen to give. That is one point
# on the axis, and the claims that matter -- the RCB partition bit-for-bit
# against its serial oracle, the native mesh's kernels, the typed numerical
# kernels -- were verified there only. `fuse_add_omp_legs` registers a test a
# second and third time, at ONE thread and at every core, so the two extremes
# are measured on every build rather than assumed from the default.
# review-2026-10-11 PS-12: this paragraph used to say the hazard these legs
# guard against is order-dependent FLOATING-POINT accumulation in
# median.hpp's histogram -- wrong, since `Weight` (types.hpp) is `int64_t`
# and `Kokkos::atomic_add` on an integer is exact and order-invariant by
# associativity; there is no FP accumulation in the histogram at all. What
# these legs actually guard against is any FUTURE order-sensitive
# accumulation this layer might grow, and they empirically PIN the bit-
# identity claim (docs/dev/partitioning.md "Determinism") at both thread-
# count extremes rather than assuming it from the default. Labelled `omp`;
# empty when Kokkos has no OpenMP.
#
# What it does NOT cover, stated so nobody reads the label as more than it is:
# the coupler's own runtime (src/runtime/, src/schedule/) executes no Kokkos
# kernel at all -- its sums and norms are the serial ones in num/kernels.hpp
# and the space kernels in num/space_kernels.hpp are verified but uncalled.
# Running the runner tests at two thread counts would compare a serial loop
# with itself. That is a capability gap (KNOWN_ISSUES.md D11), not a test gap.
# cmake/FuseHostCores.cmake: the count ignores OMP_NUM_THREADS, which
# ProcessorCount() would honour (a configure under OMP_NUM_THREADS=1 then drops
# the all-cores legs and turns every omp_identical row into a 1-vs-1 comparison).
include(FuseHostCores)
fuse_host_core_count(_fuse_host_cores)
if(_fuse_host_cores EQUAL 0)
  set(_fuse_host_cores 4)
endif()
if(_fuse_host_cores LESS 2)
  message(STATUS "fuse: ONE host core -- the omp legs and example.*.omp_identical compare one thread with one thread and cannot detect a thread-count dependence")
endif()
if(TARGET Kokkos::kokkos)
  get_target_property(_fuse_kokkos_devices Kokkos::kokkos INTERFACE_COMPILE_DEFINITIONS)
endif()

if(Kokkos_ENABLE_OPENMP OR "${Kokkos_DEVICES}" MATCHES "OPENMP")
  set(FUSE_KOKKOS_HAS_OPENMP ON)
endif()

# Device twins (docs/dev/device-resident-arcs.md step 4). A build whose
# Kokkos has a CUDA/HIP/SYCL backend registers every row passed to
# fuse_add_device_twin() a second time, as `<row>.device`, with
# FUSE_EXECUTION_SPACE=device in its environment: RunnerConfig's `Default`
# execution space reads that variable (include/fuse/runtime/runner.hpp), so
# the same binary runs its graphs under the device interpreter -- device-
# resident arcs, mirrored or device-registered fields, kernels on the device
# -- and must reproduce the host row's verdict. Labelled `device`; absent,
# not skipped, when the build has no device backend.
set(FUSE_KOKKOS_HAS_DEVICE OFF)
if(Kokkos_ENABLE_CUDA OR Kokkos_ENABLE_HIP OR Kokkos_ENABLE_SYCL
   OR "${Kokkos_DEVICES}" MATCHES "CUDA|HIP|SYCL")
  set(FUSE_KOKKOS_HAS_DEVICE ON)
endif()
set(FUSE_KOKKOS_HAS_DEVICE ${FUSE_KOKKOS_HAS_DEVICE} PARENT_SCOPE)
function(fuse_add_device_twin test_name target)
  if(NOT FUSE_KOKKOS_HAS_DEVICE)
    return()
  endif()
  cmake_parse_arguments(ARG "THROWS" "TIMEOUT" "ARGS;LABELS" ${ARGN})
  set(labels "device" ${ARG_LABELS})
  if(ARG_THROWS)
    list(APPEND labels throws)
  endif()
  add_test(NAME ${test_name}.device COMMAND ${target} ${ARG_ARGS})
  set_tests_properties(${test_name}.device PROPERTIES
    LABELS "${labels}"
    ENVIRONMENT "FUSE_EXECUTION_SPACE=device")
  if(ARG_TIMEOUT)
    set_tests_properties(${test_name}.device PROPERTIES TIMEOUT ${ARG_TIMEOUT})
  endif()
endfunction()
function(fuse_add_omp_legs test_name target)
  if(NOT FUSE_KOKKOS_HAS_OPENMP)
    return()
  endif()
  cmake_parse_arguments(ARG "THROWS" "" "ARGS" ${ARGN})
  set(labels "omp")
  # An omp leg runs the same executable as its THROWS base test, so it must
  # carry the same label: otherwise `ctest -LE throws` (tier0-no-exceptions)
  # does not exclude it and it aborts instead of failing an assertion (found
  # running that leg for the first time, CI run 36124149902, 2026-09-25).
  if(ARG_THROWS)
    list(APPEND labels throws)
  endif()
  # One core makes the list `1 1`, and a second add_test of the same name is
  # a configure error, so the duplicate is removed.
  set(_counts 1 ${_fuse_host_cores})
  list(REMOVE_DUPLICATES _counts)
  foreach(n IN LISTS _counts)
    add_test(NAME ${test_name}.omp${n} COMMAND ${target} ${ARG_ARGS})
    set_tests_properties(${test_name}.omp${n} PROPERTIES
      LABELS "${labels}"
      ENVIRONMENT "OMP_NUM_THREADS=${n};OMP_PROC_BIND=false")
  endforeach()
endfunction()
function(fuse_record_test_source rel)
  set_property(GLOBAL APPEND PROPERTY FUSE_REGISTERED_TEST_SOURCES "${rel}")
endfunction()
# fuse_tests_gated_by(<option> <rel>...) -- these sources are legitimately
# unregistered when <option> is OFF (absent, not skipped: a skipped test reads
# like a passing one). Called UNconditionally so the audit knows about them in
# every configuration.
function(fuse_tests_gated_by option)
  if(NOT ${option})
    foreach(rel IN LISTS ARGN)
      set_property(GLOBAL APPEND PROPERTY FUSE_GATED_TEST_SOURCES "${rel}")
    endforeach()
  endif()
endfunction()

function(fuse_add_gtest layer name)
  # LABELS: extra ctest labels beyond <layer>;gtest[;throws]. A test that
  # needs an optional dependency carries its name (e.g. `moab`) so the
  # whole set can be selected with `ctest -L moab` -- the first host run
  # asked exactly that and found no tests, because nothing carried it.
  cmake_parse_arguments(ARG "THROWS;NO_LAYER_LIB" "" "LIBS;LABELS" ${ARGN})
  set(target fuse_${layer}_${name}_test)
  add_executable(${target} ${layer}/${name}_test.cpp)
  fuse_record_test_source(${layer}/${name}_test.cpp)
  if(ARG_NO_LAYER_LIB)
    target_link_libraries(${target} PRIVATE fuse_test_main ${ARG_LIBS})
  else()
    target_link_libraries(${target} PRIVATE fuse::${layer} fuse_test_main ${ARG_LIBS})
  endif()
  target_compile_definitions(${target} PRIVATE FUSE_TEST_GOLDEN_DIR="${FUSE_TEST_GOLDEN_DIR}"
                                             FUSE_TEST_DATA_DIR="${FUSE_TEST_DATA_DIR}")
  add_test(NAME ${layer}.${name} COMMAND ${target})
  set(labels "${layer};gtest")
  if(ARG_THROWS)
    list(APPEND labels throws)
  endif()
  list(APPEND labels ${ARG_LABELS})
  set_tests_properties(${layer}.${name} PROPERTIES LABELS "${labels}")
endfunction()

# fuse_add_upstream_main_test(<layer> <name>) -> same path, own main(), no harness.
function(fuse_add_upstream_main_test layer name)
  set(target fuse_${layer}_${name}_test)
  add_executable(${target} ${layer}/${name}_test.cpp)
  fuse_record_test_source(${layer}/${name}_test.cpp)
  target_link_libraries(${target} PRIVATE fuse::${layer})
  add_test(NAME ${layer}.${name} COMMAND ${target})
  set_tests_properties(${layer}.${name} PROPERTIES LABELS "${layer};upstream-main")
endfunction()

# fuse_add_mpi_test(<layer> <name> [LIBS ...] [RANKS n...] [LABELS ...]) ->
# mpiexec at each rank count. Default ranks 1 2 3 4 5 (see the body): an odd
# count is included on purpose, since a bug that appears only when the ranks do
# not divide evenly is the usual kind.
# Every leg carries a timeout, not because these are slow: the failure they
# guard against IS a hang, so a regression must fail rather than stall CI.
#
# LABELS appends ctest labels beyond the `<layer>;mpi;parallel` every leg
# carries, so a row whose test needs an optional dependency can be selected
# with the others that need it -- `ctest -L moab` missed the MOAB multi-rank
# rows until this option existed (the user's host run, 2026-10-08, reported two
# MOAB rows where the build registers twelve), which is exactly the kind of
# gap that makes a selector read as coverage it does not have.
function(fuse_add_mpi_test layer name)
  cmake_parse_arguments(ARG "DEVICE_TWIN" "" "LIBS;RANKS;LABELS" ${ARGN})
  if(NOT ARG_RANKS)
    # 1 included deliberately (S5.5, build-evidence/parallel-matrix.toml row
    # A-RANKS). A single task is not a trivial case of the others: every match
    # has both ends on one rank, so the same-rank shortcut carries the whole
    # exchange and MpiTransport must not see anything at all -- the path row 7
    # pins from the other side. Tests that need peers skip themselves through
    # FUSE_REQUIRE_AT_LEAST_RANKS; one that fails here instead of skipping is
    # missing that guard, which is worth knowing.
    set(ARG_RANKS 1 2 3 4 5)
  endif()
  set(target fuse_${layer}_${name}_test)
  add_executable(${target} ${layer}/${name}_test.cpp)
  fuse_record_test_source(${layer}/${name}_test.cpp)
  target_link_libraries(${target} PRIVATE fuse_test_main ${ARG_LIBS})
  target_compile_definitions(${target} PRIVATE FUSE_TEST_GOLDEN_DIR="${FUSE_TEST_GOLDEN_DIR}"
                                             FUSE_TEST_DATA_DIR="${FUSE_TEST_DATA_DIR}")
  # REMOVE_DUPLICATES because `<layer>` is itself `parallel` for every row in
  # test/parallel/, which spelled the label list `parallel;mpi;parallel`.
  set(_fuse_mpi_labels "${layer}" mpi parallel ${ARG_LABELS})
  list(REMOVE_DUPLICATES _fuse_mpi_labels)
  foreach(n IN LISTS ARG_RANKS)
    add_test(NAME ${layer}.${name}.${n}
             COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${n} $<TARGET_FILE:${target}>)
    # Every rank is a separate process, and Kokkos::OpenMP defaults each one's
    # thread count to the WHOLE machine's core count with no awareness of the
    # other ranks sharing it -- at n=5 on an 8-core box that is 40 threads for
    # 8 cores, measured to turn a sub-second job into a 120s TIMEOUT (this
    # function's own TIMEOUT 120 below) purely from scheduler thrashing, not
    # from anything the test exercises. `fuse_add_omp_legs` above already caps
    # itself to the host's core count for the same reason; capping each rank's
    # share to max(1, cores / n) is that same discipline applied per rank.
    math(EXPR _fuse_mpi_threads "${_fuse_host_cores} / ${n}")
    if(_fuse_mpi_threads LESS 1)
      set(_fuse_mpi_threads 1)
    endif()
    # OMPI_MCA_coll_tuned_alltoallv_algorithm=2 (pairwise) works around a real
    # hang in this Open MPI (4.1.5)'s own default "basic_linear" alltoallv
    # algorithm, but ONLY on a communicator other than MPI_COMM_WORLD itself
    # (MPI_Comm_dup'd, MPI_Comm_split'd, or MPI_Comm_create_group'd -- every
    # transport/partition/search communicator this project ever builds one
    # on). Measured 2026-10-10: a bare MPI_Comm_dup followed by one
    # MPI_Alltoallv on the duplicate hangs with rank 0 returning and every
    # other rank stuck inside the collective -- reproduced with zero fuse or
    # Kokkos code involved, independent of MPI_THREAD_MULTIPLE (still hangs
    # under MPI_THREAD_FUNNELED and plain MPI_Init), independent of the BTL
    # (vader, tcp) and PML (ob1) in use, and absent entirely on MPI_COMM_WORLD
    # itself with the identical send/recv pattern. Forcing the "pairwise"
    # algorithm (`ompi_info --param coll tuned --level 9`'s own
    # coll_tuned_alltoallv_algorithm enum) made 30/30 of
    # parallel.mpi_transport's and 28/28 of parallel.comm_graph's subtests
    # pass where they previously hung; "basic_linear" (the default this
    # build's decision function was choosing for every non-WORLD communicator
    # at this rank count) could not be made to pass at all, however many
    # times it was tried. Harmless on any other MPI: an OMPI_MCA_* variable an
    # MPICH/other mpiexec does not recognise is simply unused environment, not
    # an error. If a future Open MPI release fixes this, the fix becomes a
    # no-op (pairwise is a correctness-preserving algorithm choice everywhere
    # dynamic rules reach, just not always the fastest one at huge rank
    # counts fuse does not target).
    #
    # OMPI_MCA_rmaps_base_oversubscribe=1 is unrelated: this host has 4
    # PHYSICAL cores behind its 8 logical ones (Core(s) per socket: 4,
    # `lscpu`), and Open MPI sizes its default "slot" count from physical
    # cores, not `nproc`'s 8 -- so `n=5` and up refused outright with "not
    # enough slots" even though the host plainly has 8 logical CPUs to run
    # them on. The equivalent of passing `--oversubscribe` on the `mpiexec`
    # command line; these are correctness tests at a handful of ranks, not a
    # throughput benchmark, so running more ranks than physical cores is
    # exactly the point at n=5.
    set_tests_properties(${layer}.${name}.${n} PROPERTIES TIMEOUT 120 LABELS "${_fuse_mpi_labels}"
      ENVIRONMENT "OMP_NUM_THREADS=${_fuse_mpi_threads};OMP_PROC_BIND=false;OMPI_MCA_coll_tuned_use_dynamic_rules=1;OMPI_MCA_coll_tuned_alltoallv_algorithm=2;OMPI_MCA_rmaps_base_oversubscribe=1")
  endforeach()
  # DEVICE_TWIN (docs/dev/device-resident-arcs.md Phase 3): re-registers
  # this row under FUSE_EXECUTION_SPACE=device at 1, 2 and 3 ranks --
  # several ranks sharing however many devices this build's Kokkos reports
  # (one, on higgs; fuse::abi::resolve_device_id() is what maps a local
  # rank onto one when there are fewer devices than ranks). Absent, not
  # skipped, on a build with no device backend (fuse_add_device_twin's own
  # FUSE_KOKKOS_HAS_DEVICE guard, reused here) and at a rank count this
  # row was never registered at in the first place -- a test that needs
  # RANKS 2 3 (epoch_reconcile, window_runner_mpi) gets device twins only
  # at the ranks it actually runs at, never at 1.
  if(ARG_DEVICE_TWIN AND FUSE_KOKKOS_HAS_DEVICE)
    # The intersection of {1, 2, 3} and this row's own RANKS: a row
    # registered only at RANKS 2 3 (window_runner_mpi) gets device twins
    # only at 2 and 3, never a rank count it does not run the host leg at.
    foreach(n IN ITEMS 1 2 3)
      if(NOT n IN_LIST ARG_RANKS)
        continue()
      endif()
      math(EXPR _fuse_mpi_threads "${_fuse_host_cores} / ${n}")
      if(_fuse_mpi_threads LESS 1)
        set(_fuse_mpi_threads 1)
      endif()
      add_test(NAME ${layer}.${name}.${n}.device
               COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${n} $<TARGET_FILE:${target}>)
      set_tests_properties(${layer}.${name}.${n}.device PROPERTIES TIMEOUT 120
        LABELS "${_fuse_mpi_labels};device"
        ENVIRONMENT "FUSE_EXECUTION_SPACE=device;OMP_NUM_THREADS=${_fuse_mpi_threads};OMP_PROC_BIND=false;OMPI_MCA_coll_tuned_use_dynamic_rules=1;OMPI_MCA_coll_tuned_alltoallv_algorithm=2;OMPI_MCA_rmaps_base_oversubscribe=1")
    endforeach()
  endif()
endfunction()

  # MOAB_LIBRARY_DIRS (cmake/FuseDependencies.cmake) is this project's own
  # best record of where every one of MOAB's dependencies' shared libraries
  # live, filled in from each MOAB_USE_<PKG>/<PKG>_DIR pair when MOAB's own
  # config does not already list it. It is NOT enough to rely on a MOAB
  # binary's rpath alone: this toolchain's gcc passes --as-needed on every
  # link by default (`gcc -dumpspecs`), which drops a dependency library
  # from a binary's own NEEDED list the moment nothing in THIS project's
  # sources calls its symbols directly (true of every moab_mesh_test.cpp
  # case, which never touches TempestRemap/NetCDF/PnetCDF/Zoltan directly) --
  # so resolving it at runtime falls on libMOAB.so's OWN rpath instead,
  # which is not this project's to guarantee complete. Measured 2026-10-10:
  # a MOAB rebuilt against a different dependency tree landed with a rpath
  # naming only its MPI-stack paths, breaking backend.moab_mesh and the
  # abi.install_consumer_* tests with "libTempestRemap.so.0: cannot open
  # shared object file" though nothing in this repository had changed.
  # Prepending every one of MOAB_LIBRARY_DIRS to LD_LIBRARY_PATH at test run
  # time (not baked into any one binary's own rpath, which --as-needed can
  # still defeat) makes this independent of what libMOAB.so's own rpath
  # does or does not list.
  set(_fuse_moab_env_mods "")
  foreach(_fuse_moab_libdir IN LISTS MOAB_LIBRARY_DIRS)
    list(APPEND _fuse_moab_env_mods "LD_LIBRARY_PATH=path_list_prepend:${_fuse_moab_libdir}")
  endforeach()

  fuse_add_gtest(backend moab_mesh THROWS LABELS moab)
  set_tests_properties(backend.moab_mesh PROPERTIES ENVIRONMENT_MODIFICATION "${_fuse_moab_env_mods}")

  set_tests_properties(backend.remap_halo PROPERTIES ENVIRONMENT_MODIFICATION "${_fuse_moab_env_mods}")

  # as upstream/soma/test/CMakeLists.txt discovered SOMA_MOAB_TEST_MESH (this
  # is the port of that mechanism -- FUSE_MOAB_TEST_MESH is the name the test
  # file itself already refers to; nothing previously defined it). The four
  # MoabMeshFile.* cases compile only when the file is found; its absence is
  # a configure-time message, not a test that reads as passing.
  # find_file never searches again once its variable is set, so a path into
  # an install that was later removed or replaced stayed in the cache and the
  # four MoabMeshFile.* tests compiled against a file that no longer existed
  # (measured 2026-10-13: /opt/moab-tempest removed, build-all still named

  if(FUSE_MOAB_TEST_MESH_FILE AND NOT EXISTS "${FUSE_MOAB_TEST_MESH_FILE}")
    message(STATUS
      "fuse: cached FUSE_MOAB_TEST_MESH_FILE=${FUSE_MOAB_TEST_MESH_FILE} "
      "no longer exists; searching again")
    unset(FUSE_MOAB_TEST_MESH_FILE CACHE)
  endif()
  if(MOAB_INCLUDE_DIRS)
    list(GET MOAB_INCLUDE_DIRS 0 _fuse_moab_inc)
    get_filename_component(_fuse_moab_prefix "${_fuse_moab_inc}" DIRECTORY)

              NAMES spherecube.h5m

      set(_scratch ${CMAKE_CURRENT_BINARY_DIR}/connect-scratch/${_mode}-${_na}+${_nb})

      # inherit it from this test process's environment.

        LABELS "parallel;mpi;connect" TIMEOUT 120

# Scaled device benchmark (docs/dev/device-resident-arcs.md rollout step 6
# follow-up, scaled-bench track): a native quad-mesh Remap/Transport chain
# at problem sizes orders of magnitude larger than the correctness suite,
# so per-instruction kernel-launch and host-stop overhead can be told
# apart from genuine arithmetic cost -- see bench/scaled_device_benchmark.cpp's
# own header comment for the graph and the placement/execution-space
# conventions. Not a *_test.cpp (the audit section below globs only
# *_test.cpp) and not a gtest binary -- one MPI-aware executable driven by
# different CLI arguments at each rank count and problem size, the same
# convention examples/CMakeLists.txt's own fuse_add_layout_test uses for
# layout_driver.
if(FUSE_ENABLE_MPI)
  add_executable(fuse_bench_scaled_device_benchmark bench/scaled_device_benchmark.cpp)
  target_link_libraries(fuse_bench_scaled_device_benchmark PRIVATE
    fuse::runtime fuse::backend fuse::transport MPI::MPI_CXX)

  # Small-size correctness row (RED/GREEN evidence): --verify builds and
  # runs the SAME small graph under Host and Device IN THIS ONE PROCESS
  # and asserts 1e-14 relative agreement itself -- see the file's own
  # header comment. Labelled "example": cheap (a handful of cells) and
  # belongs in the default tiers, same as mms_pair/dn_heat's own
  # correctness rows. The .device twin reruns the identical --verify
  # command under FUSE_EXECUTION_SPACE=device; --verify's own comparison
  # does not read that variable (it hardwires Host then Device inside one
  # process regardless), so the twin is a second, convention-following
  # confirmation rather than a different code path.
  add_test(NAME example.scaled_device_benchmark
           COMMAND fuse_bench_scaled_device_benchmark --verify)
  set_tests_properties(example.scaled_device_benchmark PROPERTIES
    LABELS "example" TIMEOUT 120)
  if(COMMAND fuse_add_device_twin)
    fuse_add_device_twin(example.scaled_device_benchmark fuse_bench_scaled_device_benchmark
      ARGS --verify LABELS example)
  endif()

  # The scaled rows: --n/--periods sized for a real measurement, at 1, 2
  # and 3 ranks (the disjoint provider/consumer placement degenerates to
  # "both on the whole one-rank world" at one rank -- see
  # scaled_device_benchmark.cpp's own placement_for() comment), each with a
  # FUSE_EXECUTION_SPACE=device twin at --fields device. Labelled "bench"
  # ONLY (never "example", "gtest" or "parallel"), so neither tier0's plain
  # `ctest --test-dir build` leg (.github/workflows/ci.yml, now `-LE bench`
  # on both its tier0 invocations for exactly this reason) nor a bare
  # `ctest -j1` sweeps up the scaled size by accident. Run deliberately
  # with `ctest -L bench`.
  function(fuse_add_bench_row name ranks)
    cmake_parse_arguments(ARG "" "" "ARGS" ${ARGN})
    add_test(NAME ${name}
             COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks}
                     $<TARGET_FILE:fuse_bench_scaled_device_benchmark> ${ARG_ARGS})
    set_tests_properties(${name} PROPERTIES TIMEOUT 1800 LABELS "bench;mpi;parallel")
  endfunction()

  set(FUSE_BENCH_N 1000 CACHE STRING
      "Scaled device benchmark provider grid side length (provider has N*N cells)")
  foreach(_bench_ranks 1 2 3)
    fuse_add_bench_row(bench.scaled_device_benchmark.${_bench_ranks} ${_bench_ranks}
      ARGS --n ${FUSE_BENCH_N} --periods 5 --fields host)
    if(FUSE_KOKKOS_HAS_DEVICE)
      add_test(NAME bench.scaled_device_benchmark.${_bench_ranks}.device
               COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${_bench_ranks}
                       $<TARGET_FILE:fuse_bench_scaled_device_benchmark>
                       --n ${FUSE_BENCH_N} --periods 5 --fields device)
      set_tests_properties(bench.scaled_device_benchmark.${_bench_ranks}.device PROPERTIES
        TIMEOUT 1800 LABELS "bench;mpi;parallel;device"
        ENVIRONMENT "FUSE_EXECUTION_SPACE=device")
    endif()
  endforeach()
  # KNOWN_ISSUES.md D74, FIXED (follow-up round): the nodal Remap's
  # distributed point-location search used to refuse to place the
  # domain's own corner vertex (global id 0) at 2 and 3 ranks, host and
  # device alike -- root cause was build_distributed_interp_rows's own
  # per-eval-point candidate scan testing EVERY cell in the assembled
  # mesh with no broad-phase spatial filter, so a single unrelated,
  # far-away cell's Newton non-convergence could abort an evaluation
  # point whose genuine answer the same scan had already found
  # (src/backend/nodal_remap_distributed.cpp's own comment on the fix).
  # bench.scaled_device_benchmark.{2,3}[.device] are no longer disabled;
  # confirmed green at N=1000 on this host (KNOWN_ISSUES.md D74 entry).
else()

endif()
