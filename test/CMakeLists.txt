# Test registration. Every GoogleTest binary links the harness in test/support,
# which reduces failure status across the communicator (docs/dev/testing.md).
# Tests that still carry their upstream custom main() -- the rename-only state
# of a port -- are registered as plain executables until their conversion
# commit lands. There is no third kind.

add_library(fuse_test_main STATIC support/fuse_test_main.cpp)
target_include_directories(fuse_test_main PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/support)
target_link_libraries(fuse_test_main PUBLIC fuse::base GTest::gtest)
if(FUSE_ENABLE_MPI)
  target_link_libraries(fuse_test_main PUBLIC MPI::MPI_CXX)
endif()

# fuse_add_gtest(<layer> <name> [LIBS ...]) -> test/<layer>/<name>_test.cpp
# THROWS marks a test that asserts an exception is raised; the no-exceptions CI
# leg excludes that label because with FUSE_ENABLE_EXCEPTIONS=OFF the library
# aborts instead, by contract. Membership is declared here, not detected --
# a parent tried detecting it by grep, in both directions, and both were wrong.
# Golden files are data the tests compare against, never regenerated by them.
set(FUSE_TEST_GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/golden")
set(FUSE_TEST_DATA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/data")

# `layer` names both the source directory and, by default, the library the test
# links. NO_LAYER_LIB separates them: test/verification/ is not a library layer,
# it exercises whatever the case needs (declared via LIBS).
function(fuse_add_gtest layer name)
  cmake_parse_arguments(ARG "THROWS;NO_LAYER_LIB" "" "LIBS" ${ARGN})
  set(target fuse_${layer}_${name}_test)
  add_executable(${target} ${layer}/${name}_test.cpp)
  if(ARG_NO_LAYER_LIB)
    target_link_libraries(${target} PRIVATE fuse_test_main ${ARG_LIBS})
  else()
    target_link_libraries(${target} PRIVATE fuse::${layer} fuse_test_main ${ARG_LIBS})
  endif()
  target_compile_definitions(${target} PRIVATE FUSE_TEST_GOLDEN_DIR="${FUSE_TEST_GOLDEN_DIR}"
                                             FUSE_TEST_DATA_DIR="${FUSE_TEST_DATA_DIR}")
  add_test(NAME ${layer}.${name} COMMAND ${target})
  set(labels "${layer};gtest")
  if(ARG_THROWS)
    list(APPEND labels throws)
  endif()
  set_tests_properties(${layer}.${name} PROPERTIES LABELS "${labels}")
endfunction()

# fuse_add_upstream_main_test(<layer> <name>) -> same path, own main(), no harness.
function(fuse_add_upstream_main_test layer name)
  set(target fuse_${layer}_${name}_test)
  add_executable(${target} ${layer}/${name}_test.cpp)
  target_link_libraries(${target} PRIVATE fuse::${layer})
  add_test(NAME ${layer}.${name} COMMAND ${target})
  set_tests_properties(${layer}.${name} PROPERTIES LABELS "${layer};upstream-main")
endfunction()

# ---------------------------------------------------------------- ir
# THROWS below is not decoration: these targets contain cases that assert a
# refusal is raised, and with FUSE_ENABLE_EXCEPTIONS=OFF a refusal aborts, so
# the whole target would abort in the no-exceptions leg. The label is
# target-granular, so excluding one also costs the coverage of its
# non-throwing cases -- an accepted price for now, and the reason to prefer
# per-case guards if this list grows. Determined by running the leg, not by
# reading: the nine here are exactly the targets that still aborted once the
# library stopped using throw/catch as internal control flow.

  fuse_add_gtest(ir ${t})
endforeach()
fuse_add_gtest(ir ir_dag THROWS)

# A rule enforced by a DELETED OVERLOAD can only be tested by failing to
# compile. Both targets are EXCLUDE_FROM_ALL and built only by their test, so a
# normal build is unaffected.
#
# The pair matters: WILL_FAIL alone would pass if the file failed to build for
# any reason at all -- a typo, a missing include -- so the control is the same
# shape written the permitted way and MUST build. Together they say the
# difference is the spelling.
foreach(case compile_fail_dangling_span compile_ok_owning_chain)
  add_executable(fuse_field_${case} EXCLUDE_FROM_ALL field/${case}.cpp)
  target_link_libraries(fuse_field_${case} PRIVATE fuse::base)
endforeach()

add_test(NAME field.compile_fail.dangling_span
  COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR}
          --target fuse_field_compile_fail_dangling_span)
set_tests_properties(field.compile_fail.dangling_span PROPERTIES
  WILL_FAIL TRUE LABELS "compile_fail")

add_test(NAME field.compile_ok.owning_chain
  COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR}
          --target fuse_field_compile_ok_owning_chain)
set_tests_properties(field.compile_ok.owning_chain PROPERTIES
  LABELS "compile_fail")
